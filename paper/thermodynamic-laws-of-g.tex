\section{Zeroth Law: Thermal Equilibrium Is Koru}

\begin{definition}[ThermalEquilibrium]
For a type $S$ of thermodynamic states, thermal equilibrium is an
equivalence relation $\sim$ on $S$:
\begin{enumerate}
    \item Reflexive: $\forall\, s,\; s \sim s$.
    \item Symmetric: $s \sim t \implies t \sim s$.
    \item Transitive: $s \sim t \wedge t \sim u \implies s \sim u$.
\end{enumerate}
\end{definition}

\begin{theorem}[Zeroth Law Reflexive]\label{thm:zeroth-refl}
Every state is in thermal equilibrium with itself.
Immediate from the reflexivity axiom.
\end{theorem}

\begin{theorem}[Zeroth Law Is Koru Structure]\label{thm:zeroth-koru}
The thermal equilibrium relation on an inhabited type $S$ with witness $s$
admits a Koru-compatible structure: the equivalence class of $s$ is a
web, and the koru relation (connected but distinct) holds between
distinct elements of the class.
\end{theorem}

\begin{theorem}[Zeroth Law Refutes Ender]\label{thm:zeroth-refute}
Given thermal equilibrium on $S$ with witness $s$: the ender is impossible.
\[
s : S \implies \text{AbsoluteEnder}(S) \to \bot
\]
A state in thermal equilibrium exists; existence refutes the ender. \qed
\end{theorem}

\section{First Law: Energy Conservation Is Guardian Persistence}

\begin{definition}[FirstLaw]
For a type $S$, the first law asserts energy conservation:
\begin{enumerate}
    \item $\text{energy} : S \to \mathbb{N}$ --- energy function.
    \item $\text{step} : S \to S$ --- dynamics.
    \item $\forall\, s,\; \text{energy}(\text{step}(s)) = \text{energy}(s)$ --- conservation.
\end{enumerate}
\end{definition}

\begin{theorem}[First Law Persistence]\label{thm:first-persist}
If energy is conserved and the initial state has positive energy,
then every future state has positive energy:
\[
\text{energy}(s) > 0 \implies \forall\, n,\; \text{energy}(\text{step}^n(s)) > 0
\]
\end{theorem}

\begin{theorem}[First Law Refutes Ender]\label{thm:first-refute}
Given a first law structure on $S$ with witness $s$:
$\text{AbsoluteEnder}(S) \to \bot$.

Energy cannot be destroyed. A state with energy exists. The ender
requires the type to be empty. Contradiction. \qed
\end{theorem}

\section{Second Law: Entropy Increase Is Irreversibility}

\begin{definition}[SecondLaw]
For a type $S$, the second law asserts entropy non-decrease:
\begin{enumerate}
    \item $\text{entropy} : S \to \mathbb{N}$ --- entropy function.
    \item $\text{step} : S \to S$ --- dynamics.
    \item $\forall\, s,\; \text{entropy}(\text{step}(s)) \geq \text{entropy}(s)$.
\end{enumerate}
\end{definition}

\begin{theorem}[Second Law Refutes Ender]\label{thm:second-refute}
The ender would require entropy to reach a state with no successor.
But entropy non-decrease with an inhabited type ensures the dynamics
always produce a next state. The ender has no model. \qed
\end{theorem}

\section{Third Law: Absolute Zero Is Unattainable}

\begin{definition}[ThirdLaw]
The third law asserts that absolute zero (minimum entropy state) is
unattainable in finitely many steps:
\[
\forall\, s,\; \forall\, n,\; \text{entropy}(\text{step}^n(s)) > 0
\;\text{ if }\; \text{entropy}(s) > 0
\]
\end{definition}

\begin{theorem}[Third Law Refutes Ender]\label{thm:third-refute}
If absolute zero is unattainable, the ender (which requires reaching a
state from which no continuation exists) is impossible. The system
always has a next state with positive entropy.
\end{theorem}
