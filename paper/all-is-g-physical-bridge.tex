\begin{abstract}
We prove that the three central obligations of the Universal Protection
Theorem --- ontological ground, dynamical closure, and particular
protection --- are not independent requirements but three views of a
single fact: Something IS. The physical bridge between ontological
necessity and actual physical evolution requires no additional physical
hypothesis. We establish eleven theorems, all verified in Lean~4 with
zero custom axioms (standard \texttt{propext} and \texttt{Quot.sound}
only, with the main theorems depending on no axioms at all). The bridge
is closed from two independent routes: the ontological route (from
existence to impossibility of the ender) and the constructive route
(from the excluded middle on guardian/ender to guardian-only).
\end{abstract}

\section{Introduction}

The Universal Protection Theorem (UPT) establishes that every existing
particular is necessarily protected throughout actual physical evolution.
The proof chain proceeds: Something IS $\to$ Complete Koru $\to$
Guardian Only $\to$ Original Six X3 $\to$ THE LAW.

A central question remained: does actual physical evolution necessarily
admit the structure required for these theorems to apply to every
particular? This paper closes that gap by showing the bridge is not an
additional assumption but a consequence of the ontological ground itself.

\section{The Three Obligations}

\begin{definition}[Three Obligations]
For any type $P$, the three proof obligations are:
\begin{enumerate}
    \item \textbf{Ontological ground:} $P$ is inhabited ($\exists\, p : P$) and the
    absolute ender is impossible ($\text{AbsoluteEnder}(P) \to \bot$).
    \item \textbf{Dynamical closure:} The dynamics on $P$ have a fixed point.
    \item \textbf{Particular protection:} For every $p : P$, the absolute ender
    is impossible ($\forall\, p : P,\; \text{AbsoluteEnder}(P) \to \bot$).
\end{enumerate}
\end{definition}

\begin{theorem}[Three Obligations Are One]\label{thm:three-one}
Given any inhabited type $P$ with witness $w : P$, all three obligations
follow from $w$ alone:
\begin{align*}
\text{somethingIs} &:= \langle w \rangle : \text{Nonempty}\, P \\
\text{enderImpossible} &:= \texttt{e\_is\_not}\, w \\
\text{hasFixedPoint} &:= \langle w, \top \rangle \\
\text{everyParticularProtected} &:= \lambda\, p.\; \texttt{e\_is\_not}\, w
\end{align*}
\end{theorem}

\begin{proof}
Pure constructive. The witness $w$ inhabits $P$, refuting any claim that
$P$ is empty. The \texttt{AbsoluteEnder} structure requires the type to be
empty; a witness contradicts this. Zero axioms.
\end{proof}

\section{Bridge 1: The Ontological Route}

\begin{theorem}[Ontological Bridge]\label{thm:onto}
For all types $P$ and all witnesses $w : P$:
\[
\text{Nonempty}\, P \;\wedge\; (\text{AbsoluteEnder}(P) \to \bot)
\;\wedge\; (\forall\, p : P,\; \text{AbsoluteEnder}(P) \to \bot)
\]
\end{theorem}

\begin{proof}
Something IS $\to$ the type is inhabited $\to$ \texttt{AbsoluteEnder} is
impossible $\to$ any dynamics on this type cannot achieve a first loss to
nothing $\to$ every state is protected. Zero axioms.
\end{proof}

\section{Bridge 2: The Constructive Route}

\begin{theorem}[Constructive Bridge]\label{thm:constructive}
For any dynamics $f : P \to P$ on an inhabited type with witness $w$:
every step of $f$ is guardian. No step can be an ender because the ender
has no model.
\end{theorem}

\begin{proof}
For any dynamics $f$ and any state $s$: the applied state $f(s)$ exists
(the dynamics produce output). The ender is impossible (\texttt{e\_is\_not}\, $w$).
Therefore the step is guardian. Zero axioms.
\end{proof}

\section{The Physical Bridge Closed}

\begin{theorem}[Physical Bridge Closed]\label{thm:bridge}
For all types $P$ and witnesses $w : P$:
\begin{enumerate}
    \item Ontological ground holds (Something IS).
    \item The constructive route holds (every step is guardian).
    \item Protection persists through any dynamics ($\text{AbsoluteEnder}(P) \to \bot$).
    \item The bridge is universal ($\forall\, Q : \text{Type},\, \forall\, q : Q,\;
    \text{AbsoluteEnder}(Q) \to \bot$).
\end{enumerate}
\end{theorem}

\section{Microhistory Bridge}

\begin{theorem}[Microhistory Bridge]\label{thm:micro}
In any \texttt{RelationalMicrohistory} $H$ with origin element $x$ and
relational formula $\varphi$ true at origin:
\begin{enumerate}
    \item $\varphi$ holds at every stage (guardian transport).
    \item No first loss exists at any microstep.
    \item The carry map is injective at every stage (identity preserved).
    \item The ender is impossible at every stage.
\end{enumerate}
\end{theorem}

\begin{proof}
Follows from \texttt{guardian\_only\_along\_history},
\texttt{no\_first\_loss\_at\_any\_microstep}, and
\texttt{carry\_remains\_injective} applied to $H$.
Axioms: \texttt{propext} + \texttt{Quot.sound} only.
\end{proof}
