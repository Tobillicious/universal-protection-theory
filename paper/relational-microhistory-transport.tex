\begin{abstract}
We formalize relational microhistory as a sequence of positions with
injective embeddings, indexed relations, and relational formulas evaluated
at each stage. We prove three central properties: (1)~\emph{carry injectivity}
--- the transport map between stages is injective, so no two distinct
elements are merged; (2)~\emph{intrinsic truth at every stage} --- any
relational formula true at the origin holds at every subsequent stage under
the carried element; (3)~\emph{no first loss} --- there is no microstep
where a truth held before but fails after. These results establish that
relational identity and intrinsic truth are preserved through arbitrary
discrete evolution. All theorems verified in Lean~4 with standard axioms
only (\texttt{propext} + \texttt{Quot.sound}).
\end{abstract}

\section{Introduction}

The question of persistence through change is ancient: does a thing remain
itself after transformation? We give a machine-verified answer within the
framework of relational microhistory: a discrete sequence of stages, each
carrying a set of positions, an injective embedding into the next stage,
and a family of relations.

The key insight: if the carry map is injective and the relations are
transported by the carry, then every relational truth at the origin is
preserved at every stage. This is not an assumption but a theorem.

\section{Relational Microhistory}

\begin{definition}[RelationalMicrohistory]
A relational microhistory $H$ consists of:
\begin{enumerate}
    \item A type of positions $P$.
    \item For each stage $n : \mathbb{N}$, a binary relation $R_n : P \to P \to \text{Prop}$.
    \item For each stage $n$, a step function $\sigma_n : P \to P$.
    \item A transport condition: $\forall\, n, x, y,\; R_n(x, y) \to R_{n+1}(\sigma_n(x), \sigma_n(y))$.
    \item Injectivity: $\forall\, n,\; \text{Injective}(\sigma_n)$.
\end{enumerate}
\end{definition}

\begin{definition}[Carry]
The carry map $\text{carry}(H, n) : P \to P$ is the composition of step
functions from stage 0 to stage $n$:
\[
\text{carry}(H, 0) = \text{id}, \qquad
\text{carry}(H, n+1) = \sigma_n \circ \text{carry}(H, n)
\]
\end{definition}

\section{Carry Injectivity}

\begin{theorem}[Carry Remains Injective]\label{thm:carry}
For any relational microhistory $H$ and any stage $n$:
\[
\text{Injective}(\text{carry}(H, n))
\]
\textbf{Proof.} By induction on $n$. Base: $\text{carry}(H, 0) = \text{id}$,
which is injective. Step: $\text{carry}(H, n+1) = \sigma_n \circ \text{carry}(H, n)$.
The composition of injective functions is injective. \qed

Axioms: \texttt{propext} + \texttt{Quot.sound}.
\end{theorem}

\begin{corollary}[No Merging]
Distinct elements at the origin remain distinct at every stage:
\[
\forall\, n, x, y,\; x \neq y \implies \text{carry}(H, n)(x) \neq \text{carry}(H, n)(y)
\]
\end{corollary}

\section{Intrinsic Truth at Every Stage}

\begin{definition}[Relational Formula]
A relational formula $\varphi$ over positions $P$ is evaluated against a
binary relation $R$ and a position $x$:
\[
\text{eval}(R, x, \varphi) : \text{Prop}
\]
Formulas include atomic relations, conjunction, disjunction, and negation.
\end{definition}

\begin{theorem}[Intrinsic Truth Transport]\label{thm:truth}
For any relational microhistory $H$, origin element $x : P$, and formula
$\varphi$: if $\text{eval}(R_0, x, \varphi)$ holds, then for every stage $n$:
\[
\text{eval}(R_n, \text{carry}(H, n)(x), \varphi)
\]
\textbf{Proof.} By induction on $n$, using the transport condition at each step.
The key: the transport condition moves both elements and their relations
together, so truth is preserved. \qed
\end{theorem}

\section{No First Loss}

\begin{theorem}[No First Loss at Any Microstep]\label{thm:noloss}
There is no stage $n$ where a relational truth held at stage $n$ but fails
at stage $n+1$:
\[
\neg\, \exists\, n,\; \text{eval}(R_n, \text{carry}(H, n)(x), \varphi)
\;\wedge\; \neg\, \text{eval}(R_{n+1}, \text{carry}(H, n+1)(x), \varphi)
\]
\textbf{Proof.} By contradiction. If such an $n$ existed, the transport
condition at step $n$ would move the truth from stage $n$ to stage $n+1$,
contradicting the failure. \qed
\end{theorem}

\begin{corollary}[Monotone Truth]
Relational truth is monotonically preserved: once established, never lost.
The tree of knowledge grows but never shrinks.
\end{corollary}

\section{Arbitrary Relational Duties}

\begin{theorem}[Arbitrary Duties at Every Stage]\label{thm:duties}
For any finite collection of relational formulas $\varphi_1, \ldots, \varphi_k$
all true at the origin: all hold at every stage.
\[
\bigwedge_{i=1}^{k} \text{eval}(R_0, x, \varphi_i)
\implies
\forall\, n,\; \bigwedge_{i=1}^{k} \text{eval}(R_n, \text{carry}(H, n)(x), \varphi_i)
\]
This extends truth transport to obligation bundles: if six duties hold at
the origin, all six hold at every stage.
\end{theorem}

\section{Dynamic Relational Transport}

\begin{theorem}[Forward Persistence]\label{thm:forward}
In any forward-persistent microhistory (where each step preserves the
relation structure): the carry map transports all relational content
forward without loss. Combined with injectivity, this establishes
that identity and truth are co-transported.
\end{theorem}
