Builds per-date assignments with numpy, then verifies every cap on the realized integer counts exactly.
The trace is evidence for the ledger lemma, not a replacement for it."""
import json, sys
import numpy as np
N, M, H, K, REP, C, LOST = 400, 1032, 242, 3, 73, 74, 71
SLOTS, SERVICE, PAIR, WD = 65536, 30, 1024, 8
RECV = [0, 1, 2]
LOSTI = set(RECV)

def mirrors(home): return [(home + i) % N for i in (1, 2, 3)]
def build_records(r):
    recs = []                                           # (home, rank, demand, source islands)
    srcs = [i for i in mirrors(r) if i not in LOSTI]
    for rank in range(M): recs.append((r, rank, 4 * H, srcs))
    for h in [(r - i) % N for i in (1, 2, 3)]:
        s = [i for i in [h] + mirrors(h) if i not in LOSTI and i != r]
        for rank in range(M): recs.append((h, rank, 4 * C, s))
    return recs

def schedule(recs):
    """Fair merge: virtual time (j+1/2)/d per record; returns per-date allocation matrix [date, record]."""
    d = np.array([x[2] for x in recs]); idx = np.repeat(np.arange(len(recs)), d)
    j = np.concatenate([np.arange(k) for k in d]); t = (j + 0.5) / d[idx]
    order = idx[np.argsort(t, kind="stable")]
    alloc = np.zeros((SERVICE, len(recs)), dtype=np.int64)
    for day in range(SERVICE):
        blk = order[day * SLOTS:(day + 1) * SLOTS]
        if len(blk): alloc[day] = np.bincount(blk, minlength=len(recs))
    return alloc
